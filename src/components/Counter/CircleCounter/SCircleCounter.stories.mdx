# Code

```vue
<template>
  <div id="countdown" class="relative h-10 w-10 text-center">
    <!--
      Todo Background color design conflict
    -->
    <div
      id="countdown-number"
      class="inline-block leading-[40px] text-[#334055]"
    >
      {{ number }}
    </div>
    <svg>
      <circle r="18" cx="20" cy="20" />
    </svg>
  </div>
</template>
<script setup lang="ts">
import { ref } from "vue";

interface Emits {
  (e: "finished"): void;
}

const emit = defineEmits<Emits>();

let number = ref(60);

const timer = setInterval(() => {
  if (number.value > 0) {
    number.value--;
  } else {
    clearInterval(timer);
    emit("finished");
  }
}, 1000);
</script>

<style scoped>
svg {
  position: absolute;
  top: 0;
  right: 0;
  width: 40px;
  height: 40px;
  transform: rotateY(-180deg) rotateZ(-90deg);
}

svg circle {
  stroke-dasharray: 113px;
  stroke-dashoffset: 0px;
  stroke-linecap: round;
  stroke-width: 3px;
  stroke: #334055;
  fill: none;
  animation: countdown 60s linear forwards;
}

@keyframes countdown {
  from {
    stroke-dashoffset: 0px;
  }
  to {
    stroke-dashoffset: 113px;
  }
}
</style>
```
