# Validation Password

###
A composition function for password validation in Vue 3 applications.

Returns an object with the following properties:

- password: a reactive property containing the current password value.
- passwordError: a reactive property containing an error message for the password, if any.
- validatePassword: a function for validating the password.

The password is validated against the following rules:

- It is required.
- It must be at least 8 characters long.
- It must contain at least one number.
- It must contain at least one lowercase letter.
- It must contain at least one uppercase letter.
###

```typescript

import { ref, onMounted } from 'vue'

export function usePasswordValidation() {
  const password = ref('')
  const passwordError = ref('')

  function validatePassword(value) {
    password.value = value
    passwordError.value = ''

    if (!value) {
      passwordError.value = 'Password is required'
      return
    }

    if (value.length < 8) {
      passwordError.value = 'Password must be at least 8 characters long'
      return
    }

    if (!/\d/.test(value)) {
      passwordError.value = 'Password must contain at least one number'
      return
    }

    if (!/[a-z]/.test(value)) {
      passwordError.value = 'Password must contain at least one lowercase letter'
      return
    }

    if (!/[A-Z]/.test(value)) {
      passwordError.value = 'Password must contain at least one uppercase letter'
      return
    }
  }

  onMounted(() => {
    validatePassword(password.value)
  })

  return { password, passwordError, validatePassword }
}
```

# Usage

```typescript
import { usePasswordValidation } from '/password-validation.js' // path to your file

export default {
  setup() {
    const { password, passwordError, validatePassword } = usePasswordValidation()
    return { password, passwordError, validatePassword }
  }
}
```
