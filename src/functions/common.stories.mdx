### 1. Generate dynamic routes for Sitemap

**Sitemap** can automatically generate or serve dynamic sitemap.xml for your Nuxt.js projects!

#### **1.1** Firstly you need to register Hook as module to listen Nuxt events!

```javascript
export default function () {
  const { nuxt } = this;
  nuxt.hook("generate:done", (context) => {
    const routesToExclude = []; // Add any route you don't want in your sitemap. Potentially get this from an .env file.
    const allRoutes = Array.from(context.generatedRoutes);
    const routes = allRoutes.filter(
      (route) => !routesToExclude.includes(route)
    );

    nuxt.options.sitemap.routes = [...routes];
  });
}
```

#### **1.2** Then import your module to buildModules in nuxt.config.js

```javascript
buildModules: [..."@/modules/sitemap.js"];
```

#### **1.3** Write in nuxt.config.js file, unable to verify the First Certificate in Axios Request

```javascript
import * as https from "https";
import axios from "axios";
const httpsAgent = new https.Agent({
  rejectUnauthorized: false,
});
```

#### **1.4** Then import your module to buildModules in nuxt.config.js

```javascript
sitemap: [
    {
      path: '/sitemap.xml', // Main path name
      hostname: 'https://jobhunt.uz', // Your site url
      exclude: [
        '/secret',
        '/admin/**',
        '/ru/**',
        '/uz/**',
        '/company/profile',
        '/worker/profile',
        '/worker/desk/saved',
        '/company/password',
        '/worker/password',
      ], // Non-indexable routes
      defaults: {
        changefreq: 'daily', // How frequently the page is likely to change: 'daily' | 'monthly' | 'weekly' | 'always' | 'hourly' | 'yearly'
        priority: 1, // The priority of this URL relative to other URLs on your site
        lastmod: new Date(), // The date of last modification of the page
      },
      sitemaps: [
        {
          path: '/sitemap/main.xml',
          defaults: {
            changefreq: 'daily',
            priority: 1,
            lastmod: new Date(),
          },
        },
        {
          path: '/sitemap/company.xml',
          defaults: {
            changefreq: 'daily',
            priority: 0.8,
            lastmod: new Date(),
          },
          routes: async () => {
            const company = await axios
              .get(
                'https://panel.jobhunt.uz/api/v1/company/seo/company-list/',
                {
                  httpsAgent,
                }
              )
              .then((res) =>
                res?.data?.map((user) => `/worker/company/${user?.slug}`)
              )
            if (company) {
              return company
            }
          },
        },
        {
          path: '/sitemap/vacancy.xml',
          defaults: {
            changefreq: 'daily',
            priority: 1,
            lastmod: new Date(),
          },
          routes: async () => {
            const vacancy = await axios
              .get(
                'https://panel.jobhunt.uz/api/v1/company/seo/vacancy-list/',
                {
                  httpsAgent,
                }
              )
              .then((res) =>
                res?.data?.map(
                  (user) => `/worker/vacancy/details/${user?.slug}`
                )
              )
            if (vacancy) {
              return vacancy
            }
          },
        },
      ],
    },
```

### 2. Dowload file

**Download file** is used to download file via [fetch](https://developer.mozilla.org/en-US/docs/Web/API/fetch) from the given url. You should specify url & filename. <br />
**Attention!** You should add filename and its format:

- ❌ Not: `check_file`
- ✅ Right: `check_file.pdf`

```typescript
function downloadFile(url: string, filename: string) {
  return new Promise((resolve, reject) => {
    fetch(url, {
      method: "GET",
    })
      .then((result) => {
        return result.blob();
      })
      .then((res) => {
        const url = window.URL.createObjectURL(new Blob([res]));
        const link = document.createElement("a");
        link.href = url;
        link.setAttribute("download", filename);
        link.click();
        link.remove();
        resolve(res);
      })
      .catch((err) => {
        reject(err);
      });
  });
}
```

### 3. Reset pinia stores

**Reset pinia stores** is used to reset or clear all or certain pinia stores. Example: Clear all stores: `$ResetPinia().all()`

```typescript
import { getActivePinia, Pinia, Store } from "pinia";
// ---
interface ExtendedPinia extends Pinia {
  _s: Map<string, Store>;
}
function $ResetPinia() {
  const pinia = getActivePinia() as ExtendedPinia;

  if (!pinia) {
    throw new Error("There is no stores");
  }

  const resetStores: Record<string, () => void> = {};

  pinia._s.forEach((store, name) => {
    resetStores[name] = () => store.$reset();
  });

  resetStores.all = () => pinia._s.forEach((store) => store.$reset());
  return resetStores;
}
```

### 4. Copy to clipboard

**Copy to clipboard** is used to copy the given text to the clipboard.

```typescript
async function copyText(text: string) {
  await navigator.clipboard.writeText(text);
}
```
